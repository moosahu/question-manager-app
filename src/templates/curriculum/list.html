// تبديل حالة ظهور المنهج في البوت function toggleBotVisibility(courseId, button) { $.ajax({ url: `/api/v1/courses/${courseId}/toggle-bot-visibility`, method: 'POST', data: { csrf_token: $('input[name="csrf_token"]').val() }, success: function(response) { if (response.success) { // تحديث الزر if (response.show_in_bot) { button.classList.remove('inactive'); button.classList.add('active'); button.title = 'إيقاف الظهور في البوت'; button.innerHTML = ''; } else { button.classList.remove('active'); button.classList.add('inactive'); button.title = 'تفعيل الظهور في البوت'; button.innerHTML = ''; } // تحديث الشارة const courseItem = button.closest('.course-item'); const statusBadge = courseItem.querySelector('.bot-status'); if (statusBadge) { if (response.show_in_bot) { statusBadge.classList.remove('inactive'); statusBadge.classList.add('active'); statusBadge.title = 'يظهر في البوت'; statusBadge.innerHTML = 'مفعل'; } else { statusBadge.classList.remove('active'); statusBadge.classList.add('inactive'); statusBadge.title = 'لا يظهر في البوت'; statusBadge.innerHTML = 'معطل'; } } showToast(response.message, 'success'); } else { showToast('فشل في تغيير الحالة: ' + response.error, 'danger'); } }, error: function(xhr, status, error) { showToast('خطأ في الاتصال بالخادم', 'danger'); } }); }
